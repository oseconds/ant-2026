```powershell
@'
const fs = require('fs');
const assert = require('assert/strict');
const { gunzipSync } = require('zlib');
const { createHash } = require('crypto');
const base = 'docs/ant/evidence/2026-10-07-social-graph/data/';
const manifest = JSON.parse(fs.readFileSync(base+'manifest.json'));
const digest = x => createHash('sha256').update(x).digest();
for (const seed of [20261005,20261006,20261007]) {
  const raw = JSON.parse(gunzipSync(fs.readFileSync(base+`seed${seed}-on-raw-events.json.gz`)));
  const summary = JSON.parse(fs.readFileSync(base+`seed${seed}-summary.json`));
  const movie = gunzipSync(fs.readFileSync(base+`seed${seed}-trajectory.f64.gz`));
  const hashes = gunzipSync(fs.readFileSync(base+`seed${seed}-paired-tick-hashes.bin.gz`));
  const times = [];
  for (let at=20; at<movie.length;) {
    times.push(movie.readDoubleLE(at));
    at += 12+movie.readUInt32LE(at+8)*52;
  }
  assert.equal(times.length,36001);
  assert.equal(hashes.length,times.length*manifest.pairedTickHashBytesPerFrame);
  const contacts = times.map(()=>[]), births = times.map(()=>[]);
  const tickFor = ms => {
    const seconds=ms/1000;
    let lo=0, hi=times.length;
    while (lo<hi) { const mid=(lo+hi)>>>1; if(times[mid]<seconds) lo=mid+1; else hi=mid; }
    assert.ok(lo<times.length); return lo;
  };
  for (const event of raw.contacts) contacts[tickFor(event.simulationTimeMs)].push(event);
  for (const event of raw.births) births[tickFor(event.payload.bornAtSimulationTimeMs)].push(event);
  const aggregate = {};
  for (const [field,batches] of [['contacts',contacts],['births',births]]) {
    const rows=[];
    for (let tick=0; tick<times.length; tick++) {
      const at=tick*manifest.pairedTickHashBytesPerFrame+(manifest.hashFields.length+manifest.hashFields.indexOf(field))*32;
      const row=digest(JSON.stringify(batches[tick])+'\n');
      assert.deepEqual(row,hashes.subarray(at,at+32),`${seed}:${tick}:${field}`);
      rows.push(row);
    }
    aggregate[field]=digest(Buffer.concat(rows)).toString('hex');
    assert.equal(aggregate[field],summary.graphEnabledParity[field].on);
  }
  console.log(JSON.stringify({seed,frames:times.length,contactGroups:raw.contacts.length,
    individualFeet:raw.contacts.reduce((n,e)=>n+e.contacts.length,0),
    plateBirths:raw.births.filter(e=>e.type==='plate-birth').length,birthEvents:raw.births.length,
    contactTickMismatches:0,birthTickMismatches:0,aggregate}));
}
```
